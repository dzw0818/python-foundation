```python
class ValidationError(Exception):
    def __init__(self, field, message):
        self.field = field
        self.message = message
        super().__init__(f"{field}: {message}")

class AgeError(ValidationError):
    def __init__(self, age, message):
        super().__init__("age", message)
        self.age = age

def validate_age(age):
    if not isinstance(age, int):
        raise AgeError(age, "年龄必须是整数")
    if age < 0 or age > 150:
        raise AgeError(age, "年龄必须在 0-150 之间")

# 测试
try:
    validate_age(-1)
except AgeError as e:
    print(f"字段 {e.field} 验证失败：{e.message}")
```

from datetime import datetime

def log_error(error, context=""):
    log_entry = {
        "time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "type": type(error).__name__,
        "message": str(error),
        "context": context
    }
    
    # 添加注释（Python 3.11+）
    if hasattr(error, 'add_note'):
        error.add_note(f"记录时间：{log_entry['time']}")
        error.add_note(f"上下文：{context}")
    
    print(f"[{log_entry['time']}] {log_entry['type']}: {log_entry['message']}")
    if context:
        print(f"  上下文：{context}")
    return log_entry

# 测试
errors = []

try:
    1/0
except ZeroDivisionError as e:
    errors.append(log_error(e, "计算平均值"))

try:
    int("abc")
except ValueError as e:
    errors.append(log_error(e, "用户输入处理"))

print(f"\n共记录 {len(errors)} 个错误")
